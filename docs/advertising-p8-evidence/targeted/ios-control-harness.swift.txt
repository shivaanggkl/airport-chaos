import UIKit

import UserMessagingPlatform
import Network

@MainActor func record(_ event: String, _ values: [String: Any] = [:]) {
    let url = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0].appendingPathComponent("p8-native.json")
    var events = (try? Data(contentsOf: url)).flatMap { try? JSONSerialization.jsonObject(with: $0) as? [[String: Any]] } ?? []
    var row = values; row["event"] = event; row["uptime"] = ProcessInfo.processInfo.systemUptime
    events.append(row)
    if let data = try? JSONSerialization.data(withJSONObject: events, options: [.prettyPrinted, .sortedKeys]) { try? data.write(to: url) }
    print("P8TARGET \(event) \(values)")
}
@MainActor func safeError(_ error: NSError) -> [String: Any] {
    var row: [String: Any] = ["domain": error.domain, "code": error.code, "message": error.localizedDescription]
    if let underlying = error.userInfo[NSUnderlyingErrorKey] as? NSError { row["underlying"] = ["domain": underlying.domain, "code": underlying.code, "message": underlying.localizedDescription] }
    else { row["underlying"] = NSNull() }
    if let info = error.userInfo[GADErrorUserInfoKeyResponseInfo] as? ResponseInfo {
        row["responseInfo"] = ["adapterCount": info.adNetworkInfoArray.count, "loadedAdapterPresent": info.loadedAdNetworkResponseInfo != nil,
          "adapters": info.adNetworkInfoArray.map { adapter in ["class": adapter.adNetworkClassName, "latencySeconds": adapter.latency, "error": adapter.error.map { ["domain": ($0 as NSError).domain, "code": ($0 as NSError).code, "message": $0.localizedDescription] } ?? [:]] }]
    }
    return row
}
class SceneDelegate: UIResponder, UIWindowSceneDelegate {
    var window: UIWindow?
    func scene(_ scene: UIScene, willConnectTo session: UISceneSession, options connectionOptions: UIScene.ConnectionOptions) {
        guard let scene = scene as? UIWindowScene else { return }
        window = UIWindow(windowScene: scene); window?.rootViewController = NativeControl(); window?.makeKeyAndVisible()
    }
}
class NativeControl: UIViewController, FullScreenContentDelegate {
    let status = UILabel(); var ad: RewardedAd?; var qualified = false
    let monitor = NWPathMonitor()
    override func viewDidAppear(_ animated: Bool) {
        super.viewDidAppear(animated)
        guard status.superview == nil else { return }
        view.backgroundColor = .systemBackground; status.frame = view.bounds.insetBy(dx: 30, dy: 25); status.numberOfLines = 0; status.text = "P8 native SDK control — preparing test consent"; view.addSubview(status)
        record("control-start", ["os": ProcessInfo.processInfo.operatingSystemVersionString, "mainThread": Thread.isMainThread, "appBoundDomains": Bundle.main.object(forInfoDictionaryKey: "WKAppBoundDomains") != nil])
        monitor.pathUpdateHandler = { path in Task { @MainActor in record("network-path", ["satisfied": path.status == .satisfied, "expensive": path.isExpensive]) } }; monitor.start(queue: DispatchQueue(label: "p8.network"))
        Task { @MainActor in await run() }
    }
    func run() async {
        let info = ConsentInformation.shared
        if !ProcessInfo.processInfo.arguments.contains("--restart") { info.reset() }
        let parameters = RequestParameters(); let debug = DebugSettings()
        debug.geography = ProcessInfo.processInfo.arguments.contains("--eea") ? .EEA : .other
        parameters.debugSettings = debug
        do {
            record("consent-update-start")
            try await info.requestConsentInfoUpdate(with: parameters)
            try await ConsentForm.loadAndPresentIfRequired(from: self)
            record("consent-return", ["canRequestAds": info.canRequestAds, "status": info.consentStatus.rawValue, "privacyOptions": info.privacyOptionsRequirementStatus.rawValue, "debugGeography": ProcessInfo.processInfo.arguments.contains("--eea") ? "EEA" : "OTHER"])
            guard info.canRequestAds else { status.text = "Consent SDK did not permit an ad request"; return }
            MobileAds.shared.requestConfiguration.publisherPrivacyPersonalizationState = .disabled
            MobileAds.shared.requestConfiguration.setPublisherFirstPartyIDEnabled(false)
            record("sdk-init-start"); await MobileAds.shared.start(); record("sdk-init-complete")
            let request = Request(); let extras = Extras(); extras.additionalParameters = ["npa": "1"]; request.register(extras)
            let start = ProcessInfo.processInfo.systemUptime; record("rewarded-load-start", ["canRequestAds": info.canRequestAds, "presenterActive": view.window != nil, "mainThread": Thread.isMainThread])
            do {
                ad = try await RewardedAd.load(with: "ca-app-pub-3940256099942544/1712485313", request: request)
                record("rewarded-loaded", ["durationSeconds": ProcessInfo.processInfo.systemUptime - start]); status.text = "Official test video loaded"; ad?.fullScreenContentDelegate = self
                ad?.present(from: self) { [weak self] in self?.qualified = true; record("reward-qualified") }
            } catch {
                var row = safeError(error as NSError); row["durationSeconds"] = ProcessInfo.processInfo.systemUptime - start; record("rewarded-load-error", row); status.text = "Official test ad load failed: \((error as NSError).domain):\((error as NSError).code)"
            }
        } catch { record("consent-error", safeError(error as NSError)); status.text = "Consent error: \((error as NSError).domain):\((error as NSError).code)" }
    }
    func adDidDismissFullScreenContent(_ ad: FullScreenPresentingAd) { record("dismissed", ["qualified": qualified]); self.ad = nil }
    func ad(_ ad: FullScreenPresentingAd, didFailToPresentFullScreenContentWithError error: Error) { record("presentation-error", safeError(error as NSError)); self.ad = nil }
}
