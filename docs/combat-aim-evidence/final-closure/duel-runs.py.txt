import subprocess,os,json,concurrent.futures
from pathlib import Path
out=Path('/Users/shiva/Documents/GitHub/airport-chaos/docs/combat-aim-evidence/final-closure')
def run(case):
 name,delay,lag=case
 with (out/(name+'.tap')).open('w') as f:
  r=subprocess.run(['./node_modules/.bin/tsx','--test','server/src/sky-duel.integration.test.ts'],cwd='/tmp/airport-final-closure/app',env={**os.environ,'QA_RECONNECT_DELAY':str(delay),'QA_NETWORK_DELAY':str(lag)},stdout=f,stderr=subprocess.STDOUT)
 return {'case':name,'reconnectDelay':delay,'inboundDelay':lag,'exitCode':r.returncode}
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as e:
 results=list(e.map(run,[('duel-immediate',0,0),('duel-late',3100,0),('duel-latency',1400,180)]))
(out/'duel-results.json').write_text(json.dumps(results,indent=2));print(results)
